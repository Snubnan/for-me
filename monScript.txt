    try {
      const el = document.querySelector('.b-sticky-position-dropdown');
      if (!el) return null;
      const txt = el.textContent || '';
      // match "UTC +03:00" or "UTC -05:30" or "UTC+0" etc.
      const m = txt.match(/UTC\s*([+-])\s*(\d{1,2}):?(\d{2})?/i);
      if (!m) return null;
      const sign = m[1] === '+' ? 1 : -1;
      const h = parseInt(m[2]) || 0;
      const mn = parseInt(m[3] || '0') || 0;
      return sign * (h * 60 + mn); // offset in minutes
    } catch(e){ return null; }
  };

      if (!strong) return null;
      const txt = strong.textContent.trim();
      // "Apr 6, 2026, 12:03 pm" → parse

      // d is parsed as LOCAL browser time. We need to shift to page timezone.
      const pageTZoff = getPageTZOffset(); // minutes

      // Browser local offset in minutes (note: getTimezoneOffset returns negative for east)
      const browserOff = -d.getTimezoneOffset(); // minutes east of UTC
      // Shift: d represents time in browser TZ, reinterpret as page TZ
      const shiftMs = (pageTZoff - browserOff) * 60 * 1000;
      return d.getTime() - shiftMs;
    } catch(e){ return null; }

    if (!sig.cls.length) return true;
    const c = cbCls(el);
    if (!c.length) return false;
    for (const x of c) if (sig.cls.indexOf(x) >= 0) return true;
    return false;
  };
  const cbScan = (sig, quick) => {
    const out = [];
    let nodes = [];
    try { nodes = document.querySelectorAll(sig.tag); } catch(e){ return out; }
    if (nodes.length > 12000) return out;
    for (const el of nodes) {
      if (quick && !cbQuick(sig, el)) continue;
      if (cbIsOurs(el)) continue;
      const r = cbScore(sig, el);
      if (r.sc >= 0.34) out.push({ el, sc: r.sc, kids: r.kids });
    }
    out.sort((a, b) => b.sc - a.sc);
    return out.slice(0, 14);
  };
  /* Auto-replace only for a clear winner whose inner structure matches too:
     a look-alike card from another section must never be swapped silently. */
  const CB_AUTO_MIN = 0.72, CB_AUTO_GAP = 0.10, CB_AUTO_KIDS = 0.55;
  const cbAutoPick = (c) => {
    if (!c.length) return null;
    const top = c[0];
    if (top.sc < CB_AUTO_MIN) return null;
    if (c.length > 1 && (top.sc - c[1].sc) < CB_AUTO_GAP) return null;
    if (top.kids !== null && top.kids < CB_AUTO_KIDS) return null;
    return top.el;
  };

  /* quickOnly: used by the auto-paste poller. Manual paste falls back to a full sweep. */
  const cbCandidates = (sig, quickOnly) => {
    const a = cbScan(sig, true);
    if (a.length || quickOnly) return a;
    return cbScan(sig, false);
  };

  /* ── computed-style freezing (diff against a clean sandbox default) ── */
  let _cbFrame = null;
  const _cbDef = new Map();
  const cbDefaults = (tag) => {
    if (_cbDef.has(tag)) return _cbDef.get(tag);
    let res = null;
    try {
      if (!_cbFrame || !_cbFrame.isConnected) {
        _cbFrame = document.createElement('iframe');
        _cbFrame.id = '__om_cbsandbox__';
        _cbFrame.setAttribute('aria-hidden', 'true');
        _cbFrame.style.cssText = 'position:fixed;left:-99999px;top:0;width:1280px;height:900px;border:0;opacity:0;pointer-events:none;';
        document.documentElement.appendChild(_cbFrame);
      }
      const d = _cbFrame.contentDocument;
      const el = d.createElement(tag);
      d.body.appendChild(el);
      const cs = _cbFrame.contentWindow.getComputedStyle(el);
      const o = {};
      CB_STYLE_PROPS.forEach(p => { o[p] = cs.getPropertyValue(p); });
      el.remove();
      res = o;
    } catch(e){ res = null; }
    _cbDef.set(tag, res);
    return res;
  };
  const cbFreeze = (el) => {
    let cs;
    try { cs = getComputedStyle(el); } catch(e){ return ''; }
    const def = cbDefaults(el.tagName.toLowerCase());
    const out = [];
    for (const p of CB_STYLE_PROPS) {
      let v = '';
      try { v = cs.getPropertyValue(p); } catch(e){ continue; }
      if (!v) continue;
      if (def) { if (def[p] === v) continue; }
      else if (CB_BORING.has(v)) continue;
      if (v.indexOf('blob:') >= 0) continue;
      out.push(p + ':' + v);
    }
    return out.join(';');
  };

  /* ── asset inlining ── */
  const cbBlobToData = (b) => new Promise(res => {
    try {
      const fr = new FileReader();
      fr.onload = () => res(fr.result);
      fr.onerror = () => res(null);
      fr.readAsDataURL(b);
    } catch(e){ res(null); }
  });
  const cbUrlToData = async (u) => {
    try {
      if (!u) return null;
      if (u.indexOf('data:') === 0) return u;
      const r = await fetch(u);
      if (!r.ok) return null;
      const b = await r.blob();
      if (b.size > 1800000) return null;
      return await cbBlobToData(b);
    } catch(e){ return null; }
  };

  const cbLabel = (el) => {
    const t = el.tagName.toLowerCase();
    const id = (el.id && el.id.indexOf('__om') !== 0) ? '#' + el.id : '';
    const c = cbCls(el).slice(0, 2).map(x => '.' + x).join('');
    return t + id + c;
  };
  const cbEsc = (s) => (s || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');

  /* ── COPY ── */
  const cbCopy = async (el) => {
    if (!el || !el.tagName) return;
    try {
      const els = [el, ...el.querySelectorAll('*')];
      if (els.length > 1500) { toast('✖ SUBTREE TOO BIG (' + els.length + ' nodes)'); return; }
      toast('⎘ COPYING ' + cbLabel(el) + '…');

      const clone = el.cloneNode(true);
      const cels  = [clone, ...clone.querySelectorAll('*')];
      const n = Math.min(els.length, cels.length);
      const tasks = [], canvas = [];

      for (let i = 0; i < n; i++) {
        const a = els[i], b = cels[i], tg = a.tagName;
        try { b.classList.remove('om-old-row', 'om-cb-cand', 'om-cb-act'); } catch(e){}

        if (tg === 'CANVAS') {
          const k = canvas.length; let d = null;
          try {
            d = a.toDataURL('image/png');
            if (d && d.length > 500000) {
              try { const j = a.toDataURL('image/jpeg', 0.85); if (j && j.length < d.length) d = j; } catch(e2){}
            }
            if (d && d.length > 2600000) d = null;
          } catch(e){ d = null; }   /* tainted canvas → geometry only */
          b.setAttribute('data-om-cv', String(k));
          canvas.push({ k, w: a.width, h: a.height, d });
        }

        if (tg === 'IMG') {
          const src = a.currentSrc || a.src || '';
          b.removeAttribute('srcset'); b.removeAttribute('loading');
          if (/^blob:/.test(src)) tasks.push(cbUrlToData(src).then(d => { if (d) b.setAttribute('src', d); }));
          else if (src) b.setAttribute('src', src);
        }
        if (tg === 'SOURCE') { b.removeAttribute('srcset'); b.removeAttribute('src'); }
        if (tg === 'VIDEO') {
          try {
            const cv = document.createElement('canvas');
            cv.width  = a.videoWidth  || a.clientWidth  || 320;
            cv.height = a.videoHeight || a.clientHeight || 180;
            cv.getContext('2d').drawImage(a, 0, 0, cv.width, cv.height);
            const d = cv.toDataURL('image/jpeg', 0.85);
            if (d && d.length < 2000000) b.setAttribute('poster', d);
          } catch(e){}
          b.removeAttribute('autoplay');
        }
        if (tg === 'INPUT') {
          try {
            b.setAttribute('value', a.value == null ? '' : a.value);
            if (a.checked) b.setAttribute('checked', ''); else b.removeAttribute('checked');
          } catch(e){}
        }
        if (tg === 'TEXTAREA') { try { b.textContent = a.value == null ? '' : a.value; } catch(e){} }
        if (tg === 'SELECT') {
          try {
            const oi = a.selectedIndex;
            b.querySelectorAll('option').forEach((o, k2) => {
              if (k2 === oi) o.setAttribute('selected', ''); else o.removeAttribute('selected');
            });
          } catch(e){}
        }
      }

      /* bake computed styles into the clone (index-free → survives re-parse) */
      if (CB.cfg.freeze) {
        for (let i = 0; i < n; i++) {
          const fz = cbFreeze(els[i]);
          if (!fz) continue;
          const cur = (cels[i].getAttribute('style') || '').replace(/;\s*$/, '');
          cels[i].setAttribute('style', cur ? cur + ';' + fz : fz);
        }
      }

      /* inline blob: urls left inside style attributes */
      for (let i = 0; i < n; i++) {
        const st = cels[i].getAttribute('style');
        if (!st || st.indexOf('blob:') < 0) continue;
        (st.match(/blob:[^\s"')]+/g) || []).forEach(u => {
          tasks.push(cbUrlToData(u).then(d => {
            if (!d) return;
            const cur = cels[i].getAttribute('style') || '';
            cels[i].setAttribute('style', cur.split(u).join(d));
          }));
        });
      }

      if (tasks.length) await Promise.all(tasks);

      const item = {
        id: 'cb' + Date.now().toString(36) + Math.random().toString(36).slice(2, 6),
        at: Date.now(),
        page: (location.pathname || '/').slice(0, 44),
        label: cbLabel(el),
        sig: cbSig(el),
        html: clone.outerHTML,
        canvas,
      };
      if (item.html.length > 4200000) { toast('✖ ELEMENT TOO BIG'); return; }

      CB.items.push(item);
      if (CB.items.length > 8) CB.items.shift();
      cbSave();
      cbRenderList();
      const cvMsg = canvas.length ? ' · ⬗' + canvas.length + (canvas.some(c => !c.d) ? ' (tainted!)' : '') : '';
      const clip = await cbToClipboard(item, true);
      toast('⎘ COPIED ' + item.label + cvMsg + (clip ? ' · ⧉ clipboard' : ''));
    } catch(e){ toast('✖ COPY FAILED: ' + (e && e.message ? e.message : e)); }
  };

  /* ── BUILD (deserialize + restore canvas pixels) ── */
  const cbBuild = (item) => {
    try {
      const tpl = document.createElement('template');
      tpl.innerHTML = (item.html || '').trim();
      let node = tpl.content.firstElementChild;
      if (!node) return null;
      node = document.importNode(node, true);

      const marks = [node, ...node.querySelectorAll('*')]
        .filter(x => x.hasAttribute && x.hasAttribute('data-om-cv'));
      (item.canvas || []).forEach(c => {
        const cv = marks.find(x => x.getAttribute('data-om-cv') === String(c.k));
        if (!cv) return;
        cv.removeAttribute('data-om-cv');
        if (cv.tagName !== 'CANVAS') return;
        try { if (c.w) cv.width = c.w; if (c.h) cv.height = c.h; } catch(e){}
        if (!c.d) return;
        const img = new Image();
        img.onload = () => {
          try { const g = cv.getContext('2d'); if (g) g.drawImage(img, 0, 0, cv.width, cv.height); } catch(e){}
        };
        img.src = c.d;
      });
      return node;
    } catch(e){ return null; }
  };

  /* ── keep-alive: re-insert if the SPA re-renders over our node ── */
  const cbGuard = (rec) => {
    try {
      rec.obs = new MutationObserver(() => {
        if (rec.dead || rec.node.isConnected) return;
        if (rec.url !== location.href) { rec.dead = true; try { rec.obs.disconnect(); } catch(e){} return; }
        if (rec.tries++ > 24) { rec.dead = true; try { rec.obs.disconnect(); } catch(e){} return; }
        try {
          if (rec.next && rec.next.parentNode === rec.parent) rec.parent.insertBefore(rec.node, rec.next);
          else if (rec.parent.isConnected) rec.parent.appendChild(rec.node);
          else { rec.dead = true; rec.obs.disconnect(); }
        } catch(e){ rec.dead = true; }
      });
      rec.obs.observe(document.body || document.documentElement, { childList: true, subtree: true });
    } catch(e){}
  };

  /* ── REPLACE ── */
  const cbReplace = (item, target) => {
    if (!target || !target.parentElement) { toast('✖ TARGET HAS NO PARENT'); return false; }
    const node = cbBuild(item);
    if (!node) { toast('✖ BUILD FAILED'); return false; }
    const rec = {
      node, original: target, parent: target.parentElement, next: target.nextSibling,
      itemId: item.id, url: location.href, tries: 0, dead: false, obs: null,
    };
    try { rec.parent.replaceChild(node, target); }
    catch(e){ toast('✖ REPLACE FAILED'); return false; }
    CB.pasted.push(rec);
    if (CB.cfg.keepAlive) cbGuard(rec);
    return true;
  };

  const cbRestoreAll = () => {
    cbArmStop();
    for (const r of CB.pasted.slice().reverse()) {
      r.dead = true;
      if (r.obs) { try { r.obs.disconnect(); } catch(e){} }
      try { if (r.node.isConnected) r.node.replaceWith(r.original); } catch(e){}
    }
    CB.pasted.length = 0;
  };

  /* ── PASTE ── */
  const cbPaste = (item) => {
    if (!cbGate()) return;
    const c = cbCandidates(item.sig);
    if (!c.length) { toast('✖ NO MATCH FOR ' + item.label); return; }
    const only = cbAutoPick(c);
    if (c.length === 1 && only) {
      if (cbReplace(item, only)) toast('⇄ REPLACED (' + Math.round(c[0].sc * 100) + '%)');
      return;
    }
    cbOpenPicker(item, c);
  };

  const cbPasteAll = (interactive) => {
    if (!cbGate()) return;
    if (!CB.items.length) { toast('BUFFER EMPTY'); return; }
    let ok = 0, amb = 0, none = 0, first = null;
    CB.items.slice().forEach(it => {
      const c = cbCandidates(it.sig);
      if (!c.length) { none++; return; }
      const pickEl = cbAutoPick(c);
      if (pickEl) {
        if (cbReplace(it, pickEl)) ok++; else none++;
        return;
      }
      amb++; if (!first) first = { it, c };
    });
    toast('⇄ ' + ok + ' replaced · ' + amb + ' ambiguous · ' + none + ' not found');
    if (interactive && first) cbOpenPicker(first.it, first.c);
  };

  /* ── candidate picker ── */
  const cbEnsurePicker = () => {
    if (CB.picker && CB.picker.isConnected) return CB.picker;
    const p = document.createElement('div'); p.id = '__om_cbpick__';
    const h = document.createElement('div'); h.className = 'om-pk-hdr';
    const l = document.createElement('div');
    const t = document.createElement('div'); t.className = 'om-pk-t'; t.textContent = 'Pick target';
    const s = document.createElement('div'); s.className = 'om-pk-s'; CB._pkSub = s;
    l.appendChild(t); l.appendChild(s);
    const x = document.createElement('button'); x.className = 'om-xb'; x.innerHTML = '&times;';
    x.onclick = () => cbClosePicker();
    h.appendChild(l); h.appendChild(x);
    const b = document.createElement('div'); b.className = 'om-pk-body'; CB._pkBody = b;
    p.appendChild(h); p.appendChild(b);
    document.documentElement.appendChild(p);
    CB.picker = p;
    return p;
  };
  const cbClosePicker = () => {
    if (CB.pickCands) CB.pickCands.forEach(c => { try { c.el.classList.remove('om-cb-cand', 'om-cb-act'); } catch(e){} });
    CB.pickCands = null;
    if (CB.pickBox) CB.pickBox.style.display = 'none';
    if (CB.picker)  CB.picker.style.display  = 'none';
  };
  const cbOpenPicker = (item, cands) => {
    cbClosePicker();
    const p = cbEnsurePicker();
    if (!CB.pickBox) CB.pickBox = mkHiBox('#4ade80');
    CB.pickCands = cands;
    CB._pkSub.textContent = cands.length + ' matches · ' + item.label + ' — hover to locate, click to replace';
    CB._pkBody.textContent = '';
    cands.forEach((c, i) => {
      try { c.el.classList.add('om-cb-cand'); } catch(e){}
      const r = document.createElement('div'); r.className = 'om-pk-row';
      const num = document.createElement('div'); num.className = 'om-pk-num'; num.textContent = String(i + 1);
      const tx = document.createElement('div'); tx.className = 'om-pk-tx';
      const nm = document.createElement('div'); nm.className = 'om-pk-n'; nm.textContent = cbLabel(c.el);
      const mt = document.createElement('div'); mt.className = 'om-pk-m';
      const rc = c.el.getBoundingClientRect();
      const where = cbPathOf(c.el.parentElement || c.el).slice(0, 2).reverse().join(' > ') || '—';
      mt.textContent = Math.round(rc.width) + '×' + Math.round(rc.height) +
                       (c.kids !== null && c.kids !== undefined ? ' · inner ' + Math.round(c.kids * 100) + '%' : '') +
                       ' · in ' + where +
                       ' · ' + (cbNorm(c.el.textContent).slice(0, 34) || '—');
      tx.appendChild(nm); tx.appendChild(mt);
      const sc = document.createElement('div'); sc.className = 'om-pk-sc'; sc.textContent = Math.round(c.sc * 100) + '%';
      r.appendChild(num); r.appendChild(tx); r.appendChild(sc);
      r.onmouseenter = () => {
        cands.forEach(z => { try { z.el.classList.remove('om-cb-act'); } catch(e){} });
        try { c.el.classList.add('om-cb-act'); } catch(e){}
        try { c.el.scrollIntoView({ block: 'center', behavior: 'smooth' }); } catch(e){}
        setTimeout(() => placeBox(CB.pickBox, c.el), 200);
      };
      r.onclick = () => {
        const el = c.el, sc2 = c.sc;
        cbClosePicker();
        if (cbReplace(item, el)) toast('⇄ REPLACED #' + (i + 1) + ' (' + Math.round(sc2 * 100) + '%)');
      };
      CB._pkBody.appendChild(r);
    });
    p.style.display = 'flex';
  };

  /* ── access gate ──
     The sidebar (get.js) reads the logged-in name and, if it is in
     OM_WHITELIST, puts it into the OS clipboard. Every page tab reads the
     clipboard once and unlocks the ELEMENT BUFFER for a whitelisted name.
     The verdict is cached per origin, so the clipboard is free afterwards
     for the element payloads themselves. */
  const CB_GATE_KEY = '__om_access_v1__';
  const CB_ACCESS_TAG = 'OM-ACCESS:';

  const cbNormName = (s) => (s || '').replace(/\s+/g, ' ').trim().toLowerCase();
  /* exact match only: "Niki lastx" must NOT pass as "Niki last" */
  const cbMatchName = (txt) => {
    let t = String(txt == null ? '' : txt).slice(0, 4096);
    const i = t.toUpperCase().indexOf(CB_ACCESS_TAG);
    if (i >= 0) t = t.slice(i + CB_ACCESS_TAG.length).split(/[\r\n]/)[0];
    t = cbNormName(t);
    if (!t) return null;
    for (const w of OM_WHITELIST) {
      if (cbNormName(w) === t) return w;
    }
    return null;
  };

  const cbRenderGate = () => {
    if (CB.unlocked) cbBuildSection(); else cbDropSection();
  };

  /* a cached name is re-checked against the current whitelist, so dropping a
     name from the list revokes access on the next run */
  const cbGateLoad = () => {
    if (!cbLS()) return;
    try {
      const r = localStorage.getItem(CB_GATE_KEY);
      if (!r) return;
      const o = JSON.parse(r);
      if (o && cbMatchName(o.name)) CB.unlocked = true;
      else localStorage.removeItem(CB_GATE_KEY);
    } catch(e){}
  };

  const cbUnlock = (name, quiet) => {
    CB.unlocked = true;
    if (cbLS()) { try { localStorage.setItem(CB_GATE_KEY, JSON.stringify({ name, at: Date.now() })); } catch(e){} }
    cbRenderGate();
    cbRenderList();
    if (!quiet) toast('⚿ ACCESS GRANTED · ' + name);
    cbArm();
    _cbSeen.length = 0;          /* re-read what was skipped while locked */
    cbHarvest();
  };

  const cbGate = () => CB.unlocked;   /* silent: a failed check shows nothing */


  const cbTryClipUnlock = async (interactive) => {
    let txt = '';
    try { txt = await navigator.clipboard.readText(); } catch(e){ txt = ''; }
    const hit = cbMatchName(txt);
    if (hit) { cbUnlock(hit); return true; }
    if (interactive) {
      if (txt) toast('✖ NO ALLOWED NAME IN CLIPBOARD');
      cbPasteBox('name');
    }
    return false;
  };

  /* ── the ELEMENT BUFFER section is BUILT only for a passed check ──
     While the check fails nothing of it exists in the DOM: no header,
     no buttons, no hint line. */
  let _cbHost = null, _cbAnchor = null;

  const cbDropSection = () => {
    if (CB._gateEl) { try { CB._gateEl.remove(); } catch(e){} }
    CB._gateEl = null; CB.listEl = null; CB._copyBtn = null;
  };

  const cbBuildSection = () => {
    if (!_cbHost || !_cbHost.isConnected) return;
    if (CB._gateEl && CB._gateEl.isConnected) return;

    const sec = document.createElement('div'); CB._gateEl = sec;
    sec.appendChild(mkSec('ELEMENT BUFFER'));

    const btn = document.createElement('button'); btn.className = 'om-gbt';
    btn.textContent = '⎘ copy element';
    btn.title = 'Press, then click one element on the page — it goes into the buffer';
    btn.onclick = () => cbSetCopyMode(!CB.copyMode);
    CB._copyBtn = btn;
    sec.appendChild(btn);

    sec.appendChild(mkRow('❄', 'Freeze styles', 'Bake computed CSS into the copy',
      () => CB.cfg.freeze,
      () => { CB.cfg.freeze = !CB.cfg.freeze; cbCfgSave(); toast('FREEZE STYLES ' + (CB.cfg.freeze ? 'ON' : 'OFF')); }
    ));
    sec.appendChild(mkRow('⛨', 'Keep alive', 'Re-insert if the page re-renders',
      () => CB.cfg.keepAlive,
      () => { CB.cfg.keepAlive = !CB.cfg.keepAlive; cbCfgSave(); toast('KEEP ALIVE ' + (CB.cfg.keepAlive ? 'ON' : 'OFF')); }
    ));
    sec.appendChild(mkRow('⚡', 'Auto-paste', 'Every page / route, best match',
      () => CB.cfg.autoPaste,
      () => { CB.cfg.autoPaste = !CB.cfg.autoPaste; cbCfgSave(); toast('AUTO-PASTE ' + (CB.cfg.autoPaste ? 'ON' : 'OFF')); }
    ));

    const list = document.createElement('div'); list.className = 'om-cb-list';
    CB.listEl = list; sec.appendChild(list);

    const pab = document.createElement('button'); pab.className = 'om-gbt';
    pab.textContent = '⇄ paste all (best match)';
    pab.onclick = () => cbPasteAll(true);
    sec.appendChild(pab);

    const clr = document.createElement('button'); clr.className = 'om-rbt';
    clr.textContent = '⌧ clear buffer';
    clr.onclick = () => { CB.items = []; cbSave(); cbRenderList(); toast('BUFFER CLEARED'); };
    sec.appendChild(clr);

    const hint = document.createElement('div'); hint.className = 'om-foot';
    hint.innerHTML = '⎘ pick: click = copy &nbsp;•&nbsp; ↑↓ = scope &nbsp;•&nbsp; Esc = cancel'
      + '<br>⤱ clipboard auto-collect 3s on focus &nbsp;•&nbsp; Ctrl+Alt+V = paste box';
    sec.appendChild(hint);

    sec.appendChild(mkDiv());
    _cbHost.insertBefore(sec, _cbAnchor);
    cbRenderList();
  };

  /* ── system clipboard transport ──
     localStorage only reaches documents of the same origin IN THE SAME
     browser profile. Anti-detect browsers isolate every tab (own session,
     proxy and storage), so the OS clipboard is the only channel that
     crosses tabs there. An item is packed as one JSON line. */
  const CB_MAGIC = 'OMTOOLS-ELEMENT-v1';
  const cbPack   = (item) => JSON.stringify({ m: CB_MAGIC, item });
  const cbUnpack = (txt) => {
    try {
      const o = JSON.parse((txt || '').trim());
      if (!o || o.m !== CB_MAGIC || !o.item || !o.item.html) return null;
      return o.item;
    } catch(e){ return null; }
  };

  const cbToClipboard = async (item, quiet) => {
    const txt = cbPack(item);
    if (txt.length > 6500000) { if (!quiet) toast('✖ TOO BIG FOR CLIPBOARD'); return false; }
    try {
      await navigator.clipboard.writeText(txt);
      if (!quiet) toast('⧉ CLIPBOARD ← ' + item.label + ' (' + Math.round(txt.length / 1024) + ' KB)');
      return true;
    } catch(e){}
    try {                                  /* older path, needs the click to still be "fresh" */
      const ta = document.createElement('textarea');
      ta.value = txt;
      ta.style.cssText = 'position:fixed;left:-9999px;top:0;opacity:0;';
      document.documentElement.appendChild(ta);
      ta.select(); ta.setSelectionRange(0, txt.length);
      const ok = document.execCommand('copy');
      ta.remove();
      if (ok) { if (!quiet) toast('⧉ CLIPBOARD ← ' + item.label); return true; }
    } catch(e){}
    if (!quiet) toast('✖ CLIPBOARD BLOCKED — allow it for this site');
    return false;
  };

  const cbAdopt = (item, quiet) => {
    item.id = 'cb' + Date.now().toString(36) + Math.random().toString(36).slice(2, 6);
    item.at = Date.now();
    CB.items.push(item);
    if (CB.items.length > 8) CB.items.shift();
    cbSave();
    cbRenderList();
    if (!quiet) toast('⤱ IMPORTED ' + item.label);
    cbArm();
  };

  /* manual fallback: no clipboard-read permission needed.
     mode 'item' = element payload, mode 'name' = access name from the sidebar */
  const cbPasteBox = (mode) => {
    let w = document.getElementById('__om_cbpaste__');
    if (!w) {
      w = document.createElement('div'); w.id = '__om_cbpaste__';
      const t = document.createElement('div'); t.className = 'om-pk-t';
      t.textContent = 'Press Ctrl+V here';
      const s = document.createElement('div'); s.className = 'om-pk-s';
      const ta = document.createElement('textarea');
      ta.spellcheck = false; ta.placeholder = 'paste here…';
      w.__s = s;
      const close = () => { w.style.display = 'none'; ta.value = ''; };
      const tryLoad = () => setTimeout(() => {
        const v = ta.value.trim();
        if (!v) return;
        if (w.__mode === 'name') {
          const hit = cbMatchName(v);
          if (!hit) { toast('✖ NAME NOT IN WHITELIST'); return; }
          close(); cbUnlock(hit); return;
        }
        const item = cbUnpack(v);
        if (!item) { toast('✖ NOT AN OM PAYLOAD'); return; }
        close(); cbAdopt(item);
      }, 30);
      ta.addEventListener('keydown', (ev) => {
        if (ev.key === 'Escape') { ev.preventDefault(); ev.stopPropagation(); close(); }
      }, true);
      ta.addEventListener('paste', tryLoad);
      ta.addEventListener('input', tryLoad);
      w.appendChild(t); w.appendChild(s); w.appendChild(ta);
      document.documentElement.appendChild(w);
      w.__ta = ta;
    }
    w.__mode = (mode === 'name') ? 'name' : 'item';
    w.__s.textContent = (w.__mode === 'name')
      ? 'access name copied in the sidebar · Esc to close'
      : 'element payload from the other tab · Esc to close';
    w.style.display = 'flex';
    setTimeout(() => { try { w.__ta.focus(); } catch(e){} }, 30);
  };

  const cbFromClipboard = async () => {
    if (!cbGate()) return;
    let txt = '';
    try { txt = await navigator.clipboard.readText(); } catch(e){ txt = ''; }
    const item = cbUnpack(txt);
    if (item) { cbAdopt(item); return; }
    if (txt) toast('✖ CLIPBOARD HAS NO OM PAYLOAD');
    cbPasteBox('item');
  };

  /* ── clipboard auto-collect ──
     The page can only read the CURRENT clipboard entry — Windows clipboard
     history (Win+V) is not exposed to web pages. So instead of reading a
     queue we watch the clipboard for HARVEST_MS and take everything that
     passes through it: flipping through Win+V entries makes each one
     current, and up to HARVEST_MAX distinct values get picked up. */
  const CB_HARVEST_MS   = 3000;
  const CB_HARVEST_STEP = 250;
  const CB_HARVEST_MAX  = 10;

  let _cbHarvestUntil = 0, _cbHarvestTimer = null;
  const _cbSeen = [];

  const cbSeenPush = (t) => {
    if (!t) return false;
    if (_cbSeen.indexOf(t) >= 0) return false;
    _cbSeen.push(t);
    while (_cbSeen.length > CB_HARVEST_MAX) _cbSeen.shift();
    return true;
  };

  const cbHarvestOnce = async () => {
    let txt = '';
    try { txt = await navigator.clipboard.readText(); } catch(e){ return 0; }
    if (!cbSeenPush(txt)) return 0;                       /* already handled */
    if (!CB.unlocked) {
      const hit = cbMatchName(txt);
      if (hit) cbUnlock(hit);
      return 0;
    }
    const item = cbUnpack(txt);
    if (!item) return 0;
    if (CB.items.some(z => z.label === item.label && z.html === item.html)) return 0;
    cbAdopt(item, true);
    return 1;
  };

  const cbHarvestStop = () => {
    if (_cbHarvestTimer) { clearTimeout(_cbHarvestTimer); _cbHarvestTimer = null; }
    _cbHarvestUntil = 0;
  };

  const cbHarvest = () => {
    _cbHarvestUntil = Date.now() + CB_HARVEST_MS;
    if (_cbHarvestTimer) return;                          /* running — window just extended */
    let got = 0;
    const step = () => {
      _cbHarvestTimer = null;
      cbHarvestOnce().then(n => {
        got += n;
        if (Date.now() < _cbHarvestUntil) { _cbHarvestTimer = setTimeout(step, CB_HARVEST_STEP); return; }
        if (got) toast('⤱ COLLECTED ' + got + ' FROM CLIPBOARD');
        got = 0;
      });
    };
    step();
  };

  const _cbFocusH = () => cbHarvest();
  const _cbVisH = () => { if (!document.hidden) cbHarvest(); };

  /* ── copy-mode UI ── */
  const cbHintEl = () => {
    let h = document.getElementById('__om_cbhint__');
    if (!h) { h = document.createElement('div'); h.id = '__om_cbhint__'; document.documentElement.appendChild(h); }
    return h;
  };
  const cbEnsureBox = () => { if (!CB.box || !CB.box.isConnected) CB.box = mkHiBox('#4ade80'); };
  const cbPickAt = (el) => {
    if (!el || !el.tagName) return null;
    if (el === document.body || el === document.documentElement) return null;
    try { if (el.closest(CB_OWN)) return null; } catch(e){}
    let c = el;
    for (let i = 0; i < CB.depth; i++) {
      const p = c.parentElement;
      if (!p || p === document.body || p === document.documentElement) break;
      c = p;
    }
    return c;
  };
  const cbHoverAt = (x, y) => {
    cbEnsureBox();
    const el = cbPickAt(document.elementFromPoint(x, y));
    CB.hover = el || null;
    if (!el) { CB.box.style.display = 'none'; return; }
    placeBox(CB.box, el);
    const r = el.getBoundingClientRect();
    const cv = el.querySelectorAll('canvas').length + (el.tagName === 'CANVAS' ? 1 : 0);
    cbHintEl().innerHTML = 'PICK · <b>' + cbEsc(cbLabel(el)) + '</b> · ' +
      Math.round(r.width) + '×' + Math.round(r.height) +
      (cv ? ' · ⬗' + cv : '') +
      ' · scope ' + CB.depth + ' <b>↑↓</b> · <b>Esc</b> cancel';
  };
  /* one-shot pick: armed by the button, disarms itself on the first click */
  const cbSetCopyMode = (on, quiet) => {
    if (on && !cbGate()) return;
    CB.copyMode = !!on;
    CB.depth = 0;
    const h = cbHintEl();
    if (CB.copyMode) {
      cbEnsureBox();
      h.style.display = 'block';
      h.innerHTML = 'PICK ONE ELEMENT · move the mouse over it';
      hideHiBox(); hideABox(); cancelEdit();
      cbHoverAt(S.mx, S.my);
      if (!quiet) toast('⎘ PICK AN ELEMENT — Esc to cancel');
    } else {
      h.style.display = 'none';
      if (CB.box) CB.box.style.display = 'none';
      CB.hover = null;
      if (!quiet) toast('⎘ PICK CANCELLED');
    }
    const b = CB._copyBtn;
    if (b) {
      b.textContent = CB.copyMode ? '⎘ picking… click / esc' : '⎘ copy element';
      b.classList.toggle('om-armed', CB.copyMode);
    }
  };

  /* ── buffer list inside the panel ── */
  const cbRenderList = () => {
    const L = CB.listEl;
    if (!L) return;
    L.textContent = '';
    if (!CB.items.length) {
      const e = document.createElement('div'); e.className = 'om-cb-empty';
      e.textContent = 'Buffer empty — press "copy element", click a node. Other page: the item appears here, click it. Other tab / profile: ⧉ there, ⤱ import here.';
      L.appendChild(e); return;
    }
    CB.items.slice().reverse().forEach(it => {
      const row = document.createElement('div'); row.className = 'om-cb-item';
      row.title = 'Click → find the most similar element on this page and replace it';
      const ic = document.createElement('span'); ic.className = 'om-cb-ic'; ic.textContent = '⇄';
      const tx = document.createElement('div'); tx.className = 'om-cb-tx';
      const nm = document.createElement('div'); nm.className = 'om-cb-n'; nm.textContent = it.label;
      const mt = document.createElement('div'); mt.className = 'om-cb-m';
      let tm = ''; try { tm = new Date(it.at).toLocaleTimeString(); } catch(e){}
      mt.textContent = (it.sig ? it.sig.w + '×' + it.sig.h : '?') +
        ((it.canvas && it.canvas.length) ? ' · ⬗' + it.canvas.length : '') +
        ' · ' + (it.page || '') + ' · ' + tm;
      tx.appendChild(nm); tx.appendChild(mt);
      const ex = document.createElement('div'); ex.className = 'om-cb-x om-cb-ex'; ex.textContent = '⧉';
      ex.title = 'Copy this element to the system clipboard (for another tab / profile)';
      ex.onclick = (ev) => { ev.stopPropagation(); cbToClipboard(it); };
      row.appendChild(ic); row.appendChild(tx); row.appendChild(ex);
      const x = document.createElement('div'); x.className = 'om-cb-x'; x.innerHTML = '&times;';
      x.title = 'Remove from buffer';
      x.onclick = (ev) => {
        ev.stopPropagation();
        CB.items = CB.items.filter(z => z.id !== it.id);
        cbSave(); cbRenderList();
      };
      row.appendChild(x);
      row.onclick = () => cbPaste(it);
      L.appendChild(row);
    });
  };

  /* ── navigation awareness: SPA route changes + full reloads ──
     The buffer lives in localStorage, so it survives both. On every
     route change we drop dead records, re-arm auto-paste and repair
     the panel if the site wiped it. */
  let _cbUrl = location.href;
  let _cbNavTimer = null, _cbArmTimer = null, _cbArmTries = 0, _cbAmbSaid = false;
  const _cbHist = { pushState: null, replaceState: null };

  const cbPlaced = (item) => CB.pasted.some(r => r.itemId === item.id && r.node.isConnected);

  const cbArmStop = () => { if (_cbArmTimer) { clearTimeout(_cbArmTimer); _cbArmTimer = null; } };

  /* keep trying while the route hydrates (14 passes ≈ 11 s) */
  const cbArm = () => {
    cbArmStop();
    if (!CB.unlocked) return;
    if (!CB.cfg.autoPaste || !CB.items.length) return;
    _cbArmTries = 0;
    const step = () => {
      _cbArmTimer = null;
      if (!CB.cfg.autoPaste || !CB.items.length) return;
      let hit = 0, amb = 0, left = 0;
      CB.items.slice().forEach(it => {
        if (cbPlaced(it)) return;
        left++;
        const c = cbCandidates(it.sig, true);
        if (!c.length) return;
        const pickEl = cbAutoPick(c);
      if (pickEl) {
          if (cbReplace(it, pickEl)) hit++;
        } else amb++;
      });
      if (hit) toast('⚡ AUTO-PASTE: ' + hit + ' replaced');
      if (!hit && amb && !_cbAmbSaid) { _cbAmbSaid = true; toast('⇄ ' + amb + ' ambiguous — pick in the panel'); }
      if (!left) return;
      if (++_cbArmTries < 14) _cbArmTimer = setTimeout(step, 800);
    };
    _cbArmTimer = setTimeout(step, 350);
  };

  const cbOnNav = () => {
    const u = location.href;
    if (u === _cbUrl) return;
    _cbUrl = u; _cbAmbSaid = false;
    cbClosePicker();
    cbRestoreAll();          /* leftovers must not follow us to the next route */
    cbLoad();            /* another tab may have copied something */
    cbRenderList();
    cbArm();
    cbHarvest();
  };

  const _cbNavH = () => setTimeout(cbOnNav, 0);
  /* another tab on the same origin copied / cleared something */
  const _cbStoreH = (e) => {
    if (!e || e.key !== CB_KEY) return;
    cbLoad(); cbRenderList(); cbArm();
  };

  const cbTick = () => {
    cbOnNav();
    try { if (!document.getElementById('__om_root__')) createGUI(); } catch(e){}
  };

  const cbWatchNav = () => {
    try {
      ['pushState', 'replaceState'].forEach(k => {
        const orig = history[k];
        if (!orig || orig.__om) return;
        _cbHist[k] = orig;
        const f = function(){ const r = orig.apply(this, arguments); setTimeout(cbOnNav, 0); return r; };
        f.__om = true;
        history[k] = f;
      });
    } catch(e){}
    window.addEventListener('popstate', _cbNavH, true);
    window.addEventListener('hashchange', _cbNavH, true);
    window.addEventListener('storage', _cbStoreH);
    window.addEventListener('focus', _cbFocusH);
    document.addEventListener('visibilitychange', _cbVisH);
    if (_cbNavTimer) clearInterval(_cbNavTimer);
    _cbNavTimer = setInterval(cbTick, 600);   /* routers that bypass history + panel watchdog */
  };

  /* while locked, quietly re-read the clipboard: once this origin has the
     clipboard permission the sidebar name is picked up with no interaction */
  let _cbGateTimer = null;
  const cbGateWatch = () => {
    if (_cbGateTimer) { clearInterval(_cbGateTimer); _cbGateTimer = null; }
    if (CB.unlocked) return;
    _cbGateTimer = setInterval(() => {
      if (CB.unlocked) { clearInterval(_cbGateTimer); _cbGateTimer = null; return; }
      try { cbTryClipUnlock(false); } catch(e){}
    }, 4000);
  };

  const cbInit = () => {
    cbGateLoad();
    cbRenderGate();
    if (!CB.unlocked) { try { cbTryClipUnlock(false); } catch(e){} cbGateWatch(); }
    cbWatchNav();
    cbArm();
    cbHarvest();
  };

  const cbTeardown = () => {
    CB.copyMode = false;
    cbClosePicker();
    cbArmStop();
    if (_cbGateTimer) { clearInterval(_cbGateTimer); _cbGateTimer = null; }
    if (_cbNavTimer) { clearInterval(_cbNavTimer); _cbNavTimer = null; }
    try { window.removeEventListener('popstate', _cbNavH, true); } catch(e){}
    try { window.removeEventListener('hashchange', _cbNavH, true); } catch(e){}
    try { window.removeEventListener('storage', _cbStoreH); } catch(e){}
    try { window.removeEventListener('focus', _cbFocusH); } catch(e){}
    try { document.removeEventListener('visibilitychange', _cbVisH); } catch(e){}
    cbHarvestStop();
    try { if (_cbHist.pushState) history.pushState = _cbHist.pushState; } catch(e){}
    try { if (_cbHist.replaceState) history.replaceState = _cbHist.replaceState; } catch(e){}
    for (const r of CB.pasted) { r.dead = true; if (r.obs) { try { r.obs.disconnect(); } catch(e){} } }
    CB.pasted.length = 0;
    if (CB.box)     { CB.box.remove();     CB.box = null; }
    if (CB.pickBox) { CB.pickBox.remove(); CB.pickBox = null; }
    if (CB.picker)  { CB.picker.remove();  CB.picker = null; }
    const h = document.getElementById('__om_cbhint__'); if (h) h.remove();
    const pb = document.getElementById('__om_cbpaste__'); if (pb) pb.remove();
    if (_cbFrame) { try { _cbFrame.remove(); } catch(e){} _cbFrame = null; }
    _cbDef.clear();
  };

  cbLoad(); cbCfgLoad();


  /* ════════════════════════════════════════════
     GUI BUILDER
  ════════════════════════════════════════════ */
  const mkSw=(getState)=>{
    const tr=document.createElement('div'); tr.className=`om-sw ${getState()?'on':'off'}`;
    const k=document.createElement('div'); k.className='om-sw-k'; tr.appendChild(k);
    tr.refresh=()=>{ tr.className=`om-sw ${getState()?'on':'off'}`; };
    return tr;
  };

  const mkRow=(ico,lbl,sub,getState,onToggle)=>{
    const row=document.createElement('div'); row.className='om-row';
    const left=document.createElement('div'); left.className='om-row-l';
    const ic=document.createElement('span'); ic.className='om-ico'; ic.textContent=ico;
    const tx=document.createElement('div'); tx.className='om-txt';
    const lb=document.createElement('div'); lb.className='om-lbl'; lb.textContent=lbl; tx.appendChild(lb);
    if(sub){const sb=document.createElement('div');sb.className='om-sub';sb.textContent=sub;tx.appendChild(sb);}
    left.appendChild(ic);left.appendChild(tx);
    const sw=mkSw(getState);
    row.appendChild(left);row.appendChild(sw);
    row.onclick=()=>{onToggle();sw.refresh();};
    return row;
  };

  const mkDiv=()=>{ const d=document.createElement('div');d.className='om-div';return d; };
  const mkSec=(t)=>{ const d=document.createElement('div');d.className='om-sec';d.textContent=t;return d; };

  let _hlRowEl = null; // ref to hl row for enable/disable

  const createGUI=()=>{
    if(document.getElementById('__om_root__'))return;
    injectStyles();

    const root=document.createElement('div'); root.id='__om_root__';

    /* tab */
    const tab=document.createElement('div'); tab.id='__om_tab__';
    const ti=document.createElement('span'); ti.className='om-ti'; ti.textContent='◈';
    const tl=document.createElement('span'); tl.className='om-tl'; tl.textContent='Tools';
    tab.appendChild(ti);tab.appendChild(tl);

    /* wrap */
    const wrap=document.createElement('div'); wrap.id='__om_wrap__';

    /* panel */
    const panel=document.createElement('div'); panel.id='__om_panel__';

    /* header */
    const hdr=document.createElement('div'); hdr.className='om-hdr';
    const hl=document.createElement('div'); hl.className='om-hdr-l';
    const dot=document.createElement('div'); dot.className='om-dot';
    const tt=document.createElement('span'); tt.className='om-title-t'; tt.textContent='OM';
    const ts=document.createElement('span'); ts.className='om-title-s'; ts.textContent='Tools';
    const vt=document.createElement('span'); vt.className='om-ver-tag'; vt.textContent=`v${VERSION}`;
    hl.appendChild(dot);hl.appendChild(tt);hl.appendChild(ts);hl.appendChild(vt);
    const xb=document.createElement('button'); xb.className='om-xb'; xb.innerHTML='&times;'; xb.onclick=()=>togglePanel(false);
    hdr.appendChild(hl);hdr.appendChild(xb);
    panel.appendChild(hdr);

    /* body */
    const body=document.createElement('div'); body.className='om-body';

    // ─ EDITOR ─
    body.appendChild(mkSec('EDITOR'));
    body.appendChild(mkRow('✏','Ctrl Editor','Hold Ctrl + click text',
      ()=>S.ctrlEditor,
      ()=>{ S.ctrlEditor=!S.ctrlEditor; if(!S.ctrlEditor){hideHiBox();cancelEdit();} toast(`CTRL EDITOR ${S.ctrlEditor?'ON':'OFF'}`); }
    ));
    body.appendChild(mkRow('✗','Alt Remover','Hold Alt + click element',
      ()=>S.altRemover,
      ()=>{ S.altRemover=!S.altRemover; if(!S.altRemover){hideABox();AS.down=false;} toast(`ALT REMOVER ${S.altRemover?'ON':'OFF'}`); }
    ));

    body.appendChild(mkDiv());

    // ─ SEARCH ─
    body.appendChild(mkSec('SEARCH'));
    const badge=document.createElement('div'); badge.className='om-badge';
    badge.innerHTML='<span>⌕</span><span>F3 — open / close search</span>';
    body.appendChild(badge);

    body.appendChild(mkDiv());

    // ─ FEED FILTER ─
    body.appendChild(mkSec('FEED FILTER'));

    const hideRow = mkRow('◎','Hide non-#ad','/statistics/engagement only',
      ()=>S.hideNonAd,
      ()=>{
        S.hideNonAd=!S.hideNonAd;
        if(S.hideNonAd) startHide(); else { stopHide(); S.hlOld=false; if(_hlRowEl)_hlRowEl.classList.add('om-disabled'); }
        // sync highlight row state
        if(_hlRowEl){ if(S.hideNonAd)_hlRowEl.classList.remove('om-disabled'); else _hlRowEl.classList.add('om-disabled'); }

        toast(`HIDE NON-AD ${S.hideNonAd?'ON':'OFF'}`);
      }

    const hlRow = mkRow('🔴','Highlight >24h','Requires Hide non-#ad ON',
      ()=>S.hlOld,
      ()=>{
        if(!S.hideNonAd)return; // guard
        S.hlOld=!S.hlOld;
        applyHide(); // re-run to add/remove highlights
        toast(`HIGHLIGHT OLD ${S.hlOld?'ON':'OFF'}`);
      }
    );
    _hlRowEl = hlRow;
    hlRow.classList.add('om-disabled'); // disabled by default (hideNonAd is off)
    const hlSw = hlRow.querySelector('.om-sw');
    body.appendChild(hlRow);

    body.appendChild(mkDiv());

    /* ELEMENT BUFFER section is inserted here, but only once the check passes */
    const rb=document.createElement('button'); rb.className='om-rbt'; rb.textContent='↩ RESET ALL CHANGES'; rb.onclick=resetAll;
    body.appendChild(rb);
    _cbHost = body; _cbAnchor = rb;
    cbRenderGate();

    const ft=document.createElement('div'); ft.className='om-foot';
    ft.innerHTML='Ctrl+Alt+⌫ = reset all &nbsp;•&nbsp; F3 = search';
    body.appendChild(ft);

    panel.appendChild(body);
    wrap.appendChild(panel);
    root.appendChild(wrap);
    root.appendChild(tab);
    document.documentElement.appendChild(root);

    tab.addEventListener('click',()=>togglePanel());
  };

  const togglePanel=(force)=>{
    const w=document.getElementById('__om_wrap__'); if(!w)return;
    const open=force!==undefined?force:!w.classList.contains('open');
    if(open)w.classList.add('open'); else w.classList.remove('open');
  };

  /* ════════════════════════════════════════════
     EVENTS
  ════════════════════════════════════════════ */
  const onKD=(e)=>{
    if(e.key==='F3'){e.preventDefault();e.stopPropagation();toggleSearch();return;}
    if(CB.copyMode){
      if(e.key==='Escape'){e.preventDefault();e.stopPropagation();cbSetCopyMode(false);return;}
      if(e.key==='ArrowUp'||e.key==='ArrowDown'){
        e.preventDefault();e.stopPropagation();
        CB.depth=Math.max(0,CB.depth+(e.key==='ArrowUp'?1:-1));
        cbHoverAt(S.mx,S.my);return;
      }
    }
    if(e.key==='Escape'&&CB.picker&&CB.picker.style.display!=='none'){
      e.preventDefault();e.stopPropagation();cbClosePicker();return;
    }
    if(e.ctrlKey&&e.altKey&&e.key==='Backspace'){e.preventDefault();e.stopPropagation();resetAll();return;}
    if(e.ctrlKey&&e.altKey&&(e.key==='k'||e.key==='K')){e.preventDefault();e.stopPropagation();cbTryClipUnlock(true);return;}
    if(e.ctrlKey&&e.altKey&&(e.key==='v'||e.key==='V')){e.preventDefault();e.stopPropagation();if(CB.unlocked)cbPasteBox('item');return;}
    if(e.key==='Control'&&S.ctrlEditor){S.ctrlDown=true;ensureHiBox();}
    if(e.key==='Alt'&&!AS.down&&S.altRemover){
      AS.down=true;ensureABox();
      const el=document.elementFromPoint(S.mx,S.my);
      const tg=el&&el.closest(ALT_SEL); AS.hovered=tg||null;
      if(tg)placeBox(AS.hiBox,tg);else hideABox();
    }
  };
  const onKU=(e)=>{
    if(e.key==='Control'){S.ctrlDown=false;hideHiBox();}
    if(e.key==='Alt'){AS.down=false;hideABox();AS.hovered=null;}
  };
  const onMM=(e)=>{
    S.mx=e.clientX;S.my=e.clientY;
    if(CB.copyMode){cbHoverAt(e.clientX,e.clientY);return;}
    if(S.ctrlEditor&&S.ctrlDown&&!S.editor){ensureHiBox();placeBox(S.hiBox,pickTarget(document.elementFromPoint(e.clientX,e.clientY))||document.elementFromPoint(e.clientX,e.clientY));}
    if(S.altRemover&&AS.down){const tg=document.elementFromPoint(e.clientX,e.clientY)?.closest(ALT_SEL);AS.hovered=tg||null;if(tg)placeBox(AS.hiBox,tg);else hideABox();}
  };
  const onMD=(e)=>{
    if(CB.copyMode){
      let own=false; try{ own=!!(e.target&&e.target.closest&&e.target.closest('#__om_root__,#__om_cbpick__')); }catch(err){}
      if(own)return;
      e.preventDefault();e.stopPropagation();
      const el=CB.hover||cbPickAt(document.elementFromPoint(e.clientX,e.clientY));
      if(el){ cbSetCopyMode(false,true); cbCopy(el); }
      else toast('✖ NOTHING UNDER CURSOR');
      return;
    }
    if(S.altRemover&&AS.down&&AS.hovered){
